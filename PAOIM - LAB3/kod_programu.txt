package lab3.model;

import lab3.exception.InvalidDataException;
import lab3.exception.InvalidPriceException;

import java.time.Year;
import java.util.Locale;
import java.util.Objects;

/**
 * Laptop przyjęty do odsprzedaży.
 * <p>
 * Porządek naturalny ({@link Comparable}) i równość biznesowa obejmują model, markę i wiek
 * (wynikający z roku produkcji). Dwa egzemplarze o tym samym modelu, marce i roku są traktowane
 * jako ten sam laptop przy dodawaniu do oddziału.
 */
public class Laptop implements Comparable<Laptop> {
    private static int nextSerial = 1;

    private final String model;
    private final String brand;
    private LaptopCondition condition;
    private int year;
    private double price;
    private final int ramGb;
    private final String processor;
    private final int batteryHealth;
    private final String serialNumber;

    /**
     * Klucz używany wyłącznie przy wyszukiwaniu binarnym po modelu.
     * Nie przechodzi walidacji pełnego egzemplarza.
     */
    private Laptop(String model) {
        this.model = model;
        this.brand = "";
        this.condition = LaptopCondition.WORKING;
        this.year = Year.now().getValue();
        this.price = 0;
        this.ramGb = 0;
        this.processor = "";
        this.batteryHealth = 0;
        this.serialNumber = "";
    }

    public Laptop(String model, String brand, LaptopCondition condition, int year, double price)
            throws InvalidPriceException, InvalidDataException {
        this(model, brand, condition, year, price, 8, "niepodany", 100, nextSerial());
    }

    public Laptop(String model, String brand, LaptopCondition condition, int year, double price,
                  int ramGb, String processor, int batteryHealth)
            throws InvalidPriceException, InvalidDataException {
        this(model, brand, condition, year, price, ramGb, processor, batteryHealth, nextSerial());
    }

    public Laptop(String model, String brand, LaptopCondition condition, int year, double price,
                  int ramGb, String processor, int batteryHealth, String serialNumber)
            throws InvalidPriceException, InvalidDataException {
        this.model = requireText(model, "Model");
        this.brand = requireText(brand, "Marka");
        if (condition == null) {
            throw new InvalidDataException("Stan laptopa jest wymagany.");
        }
        checkYear(year);
        validatePrice(price);
        if (ramGb <= 0) {
            throw new InvalidDataException("Pamięć RAM musi być dodatnia.");
        }
        this.processor = requireText(processor, "Procesor");
        if (batteryHealth < 0 || batteryHealth > 100) {
            throw new InvalidDataException("Kondycja baterii musi być w zakresie 0-100%.");
        }
        this.serialNumber = requireText(serialNumber, "Numer seryjny");
        this.condition = condition;
        this.year = year;
        this.price = price;
        this.ramGb = ramGb;
        this.batteryHealth = batteryHealth;
    }

    static Laptop modelKey(String model) {
        return new Laptop(model);
    }

    public void print() {
        System.out.println("Laptop: " + brand + " " + model);
        System.out.println("  Numer seryjny: " + serialNumber);
        System.out.println("  Stan: " + condition);
        System.out.println("  Rok produkcji: " + year + " (wiek: " + age() + " lat)");
        System.out.printf("  Cena: %.2f zł%n", price);
        System.out.println("  RAM: " + ramGb + " GB");
        System.out.println("  Procesor: " + processor);
        System.out.println("  Kondycja baterii: " + batteryHealth + "%");
    }

    /**
     * Porównanie według modelu, marki, a następnie wieku (starszy laptop jest „większy”).
     */
    @Override
    public int compareTo(Laptop other) {
        int byModel = model.compareTo(other.model);
        if (byModel != 0) {
            return byModel;
        }
        int byBrand = brand.compareTo(other.brand);
        if (byBrand != 0) {
            return byBrand;
        }
        return Integer.compare(age(), other.age());
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (!(obj instanceof Laptop)) {
            return false;
        }
        Laptop other = (Laptop) obj;
        return year == other.year && model.equals(other.model) && brand.equals(other.brand);
    }

    @Override
    public int hashCode() {
        return Objects.hash(model, brand, year);
    }

    @Override
    public String toString() {
        return String.format("%s %s, rok %d (wiek %d), %s, %.2f zł, nr %s",
                brand, model, year, age(), condition, price, serialNumber);
    }

    public int age() {
        return Year.now().getValue() - year;
    }

    public String getModel() {
        return model;
    }

    public String getBrand() {
        return brand;
    }

    public LaptopCondition getCondition() {
        return condition;
    }

    public int getYear() {
        return year;
    }

    public double getPrice() {
        return price;
    }

    public String getPriceLabel() {
        return String.format(Locale.US, "%.2f zł", price);
    }

    public int getRamGb() {
        return ramGb;
    }

    public String getProcessor() {
        return processor;
    }

    public int getBatteryHealth() {
        return batteryHealth;
    }

    public String getSerialNumber() {
        return serialNumber;
    }

    public void setCondition(LaptopCondition condition) throws InvalidDataException {
        if (condition == null) {
            throw new InvalidDataException("Stan laptopa jest wymagany.");
        }
        this.condition = condition;
    }

    public void setPrice(double price) throws InvalidPriceException {
        validatePrice(price);
        this.price = price;
    }

    void setYear(int year) throws InvalidDataException {
        checkYear(year);
        this.year = year;
    }

    static void checkYear(int year) throws InvalidDataException {
        int current = Year.now().getValue();
        if (year < 1980 || year > current) {
            throw new InvalidDataException(
                    "Nieprawidłowy rok produkcji: " + year + " (dozwolone 1980-" + current + ").");
        }
    }

    private static void validatePrice(double price) throws InvalidPriceException {
        if (Double.isNaN(price) || Double.isInfinite(price)) {
            throw new InvalidPriceException("Cena ma nieprawidłową wartość.");
        }
        if (price < 0) {
            throw new InvalidPriceException("Cena nie może być ujemna.");
        }
    }

    private static String requireText(String value, String field) throws InvalidDataException {
        if (value == null || value.trim().isEmpty()) {
            throw new InvalidDataException(field + " nie może być pusty.");
        }
        return value.trim();
    }

    private static synchronized String nextSerial() {
        return String.format("SN-%05d", nextSerial++);
    }
}

package lab3.model;

import lab3.exception.BranchFullException;
import lab3.exception.DuplicateLaptopException;
import lab3.exception.InvalidDataException;
import lab3.exception.InvalidPriceException;
import lab3.exception.LaptopNotFoundException;

import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;
import java.util.Locale;

/**
 * Oddział (punkt sprzedaży) przechowujący laptopy do odsprzedaży.
 * Lista jest zadeklarowana jako {@link List}, a konkretną implementacją jest {@link ArrayList}:
 * oddziały są niewielkie, często czytamy je po indeksie i sortujemy kopie.
 */
public class Branch {
    private final String branchName;
    private final List<Laptop> laptopList;
    private final int maxCapacity;

    public Branch(String branchName, int maxCapacity) throws InvalidDataException {
        if (branchName == null || branchName.trim().isEmpty()) {
            throw new InvalidDataException("Nazwa oddziału jest wymagana.");
        }
        if (maxCapacity <= 0) {
            throw new InvalidDataException("Pojemność oddziału musi być dodatnia.");
        }
        this.branchName = branchName.trim();
        this.maxCapacity = maxCapacity;
        this.laptopList = new ArrayList<>();
    }

    /**
     * Dodaje laptop. Ten sam model, marka i rok są duplikatem.
     * Przekroczenie pojemności zgłasza {@link BranchFullException}.
     */
    public void addLaptop(Laptop laptop)
            throws DuplicateLaptopException, BranchFullException, InvalidDataException {
        if (laptop == null) {
            throw new InvalidDataException("Nie można dodać laptopa do oddziału \"" + branchName + "\": brak obiektu.");
        }
        if (laptopList.contains(laptop)) {
            throw new DuplicateLaptopException("Laptop już istnieje w oddziale \"" + branchName
                    + "\" (ten sam model, marka i wiek): " + laptop.getBrand() + " " + laptop.getModel()
                    + ", rok " + laptop.getYear() + ".");
        }
        if (laptopList.size() >= maxCapacity) {
            throw new BranchFullException("Oddział \"" + branchName + "\" jest pełny ("
                    + laptopList.size() + "/" + maxCapacity + "). Nie dodano: "
                    + laptop.getBrand() + " " + laptop.getModel() + ".");
        }
        laptopList.add(laptop);
    }

    /** Usuwa laptop z oddziału (utylizacja albo zwrot do dostawcy). */
    public Laptop removeLaptop(Laptop laptop) throws LaptopNotFoundException, InvalidDataException {
        Laptop removed = take(laptop);
        if (removed == null) {
            throw new LaptopNotFoundException(missingLaptopMessage());
        }
        return removed;
    }

    /** Usuwa laptop w wyniku sprzedaży. */
    public Laptop sellLaptop(Laptop laptop) throws LaptopNotFoundException, InvalidDataException {
        Laptop sold = take(laptop);
        if (sold == null) {
            throw new LaptopNotFoundException("Nie sprzedano — w oddziale \"" + branchName + "\" nie ma takiego laptopa.");
        }
        return sold;
    }

    public void changeCondition(Laptop laptop, LaptopCondition condition)
            throws LaptopNotFoundException, InvalidDataException {
        if (condition == null) {
            throw new InvalidDataException("Nie zmieniono stanu: brak nowej wartości.");
        }
        Laptop stored = find(laptop);
        if (stored == null) {
            throw new LaptopNotFoundException(missingLaptopMessage());
        }
        stored.setCondition(condition);
    }

    public void changePrice(Laptop laptop, double price)
            throws LaptopNotFoundException, InvalidPriceException, InvalidDataException {
        Laptop stored = find(laptop);
        if (stored == null) {
            throw new LaptopNotFoundException(missingLaptopMessage());
        }
        stored.setPrice(price);
    }

    public void changeYear(Laptop laptop, int year)
            throws LaptopNotFoundException, InvalidDataException, DuplicateLaptopException {
        int index = indexOf(laptop);
        if (index < 0) {
            throw new LaptopNotFoundException(missingLaptopMessage());
        }
        Laptop stored = laptopList.get(index);
        if (stored.getYear() == year) {
            return;
        }
        Laptop.checkYear(year);
        for (int i = 0; i < laptopList.size(); i++) {
            if (i == index) {
                continue;
            }
            Laptop other = laptopList.get(i);
            if (other.getModel().equals(stored.getModel())
                    && other.getBrand().equals(stored.getBrand())
                    && other.getYear() == year) {
                throw new DuplicateLaptopException("Nie zmieniono roku " + stored.getBrand() + " " + stored.getModel()
                        + ": w oddziale \"" + branchName + "\" jest już taki model, marka i wiek (rok " + year + ").");
            }
        }
        stored.setYear(year);
    }

    public int countByCondition(LaptopCondition condition) {
        if (condition == null) {
            return 0;
        }
        int count = 0;
        for (Laptop laptop : laptopList) {
            if (laptop.getCondition() == condition) {
                count++;
            }
        }
        return count;
    }

    /** Nowa lista posortowana alfabetycznie po modelu. Kolejność w oddziale zostaje bez zmian. */
    public List<Laptop> sortByName() {
        List<Laptop> sorted = new ArrayList<>(laptopList);
        Collections.sort(sorted, Comparator.comparing(Laptop::getModel));
        return sorted;
    }

    /** Nowa lista posortowana po cenie rosnąco. */
    public List<Laptop> sortByPrice() {
        List<Laptop> sorted = new ArrayList<>(laptopList);
        Collections.sort(sorted, new Comparator<Laptop>() {
            @Override
            public int compare(Laptop o1, Laptop o2) {
                return Double.compare(o1.getPrice(), o2.getPrice());
            }
        });
        return sorted;
    }

    /**
     * Zwraca laptop o podanym modelu albo {@code null}.
     * Lista jest sortowana komparatorem po modelu, a wynik znajduje {@code Collections.binarySearch}.
     */
    public Laptop search(String model) {
        if (model == null || laptopList.isEmpty()) {
            return null;
        }
        List<Laptop> sorted = new ArrayList<>(laptopList);
        Comparator<Laptop> byModel = Comparator.comparing(Laptop::getModel);
        Collections.sort(sorted, byModel);
        int index = Collections.binarySearch(sorted, Laptop.modelKey(model), byModel);
        if (index < 0) {
            return null;
        }
        return sorted.get(index);
    }

    /** Wszystkie laptopy, których model albo marka zawiera podany fragment ({@code String.contains}). */
    public List<Laptop> searchPartial(String text) {
        List<Laptop> found = new ArrayList<>();
        if (text == null || text.isEmpty()) {
            return found;
        }
        for (Laptop laptop : laptopList) {
            if (laptop.getModel().contains(text) || laptop.getBrand().contains(text)) {
                found.add(laptop);
            }
        }
        return found;
    }

    public void summary() {
        System.out.println("--- Oddział: " + branchName + " (" + laptopList.size() + "/" + maxCapacity + ") ---");
        if (laptopList.isEmpty()) {
            System.out.println("Brak laptopów.");
            return;
        }
        int index = 1;
        for (Laptop laptop : laptopList) {
            System.out.println(index + ". " + laptop);
            index++;
        }
        System.out.printf("Łączna wartość: %.2f zł%n", totalValue());
    }

    /** Najdroższy laptop w oddziale. Pusta półka zwraca {@code null}. */
    public Laptop max() {
        if (laptopList.isEmpty()) {
            return null;
        }
        return Collections.max(laptopList, new Comparator<Laptop>() {
            @Override
            public int compare(Laptop o1, Laptop o2) {
                return Double.compare(o1.getPrice(), o2.getPrice());
            }
        });
    }

    public double totalValue() {
        double sum = 0;
        for (Laptop laptop : laptopList) {
            sum += laptop.getPrice();
        }
        return sum;
    }

    public boolean contains(Laptop laptop) {
        return laptop != null && laptopList.contains(laptop);
    }

    public boolean isEmpty() {
        return laptopList.isEmpty();
    }

    public boolean isFull() {
        return laptopList.size() >= maxCapacity;
    }

    public int size() {
        return laptopList.size();
    }

    public String getBranchName() {
        return branchName;
    }

    public int getMaxCapacity() {
        return maxCapacity;
    }

    public int getLaptopCount() {
        return laptopList.size();
    }

    public double fillRatio() {
        return (double) laptopList.size() / maxCapacity;
    }

    public String getFillLabel() {
        return String.format(Locale.US, "%.1f%%", fillRatio() * 100.0);
    }

    public List<Laptop> getLaptops() {
        return Collections.unmodifiableList(laptopList);
    }

    /** Wyjmuje egzemplarz z półki i zwraca obiekt przechowywany w oddziale. */
    Laptop take(Laptop laptop) throws InvalidDataException {
        int index = indexOf(laptop);
        if (index < 0) {
            return null;
        }
        return laptopList.remove(index);
    }

    /** Wkłada laptop już sprawdzony przez wywołującego (używane przy przeniesieniu). */
    void attach(Laptop laptop) {
        laptopList.add(laptop);
    }

    private Laptop find(Laptop laptop) throws InvalidDataException {
        int index = indexOf(laptop);
        if (index < 0) {
            return null;
        }
        return laptopList.get(index);
    }

    private int indexOf(Laptop laptop) throws InvalidDataException {
        if (laptop == null) {
            throw new InvalidDataException("Nie wskazano laptopa.");
        }
        return laptopList.indexOf(laptop);
    }

    private String missingLaptopMessage() {
        return "Brak laptopa w oddziale \"" + branchName + "\".";
    }
}

package lab3.model;

import lab3.exception.BranchFullException;
import lab3.exception.BranchNotFoundException;
import lab3.exception.DuplicateBranchException;
import lab3.exception.DuplicateLaptopException;
import lab3.exception.InvalidDataException;
import lab3.exception.LaptopNotFoundException;

import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

/**
 * Sieć oddziałów. Kluczem mapy jest nazwa oddziału.
 * {@link LinkedHashMap} zachowuje kolejność dodania przy podglądzie po nazwie,
 * a lista {@link #branches} jest tą samą referencją, którą sortuje interfejs.
 */
public class StoreManager {
    private final List<Branch> branches = new ArrayList<>();
    private final List<Branch> branchView = Collections.unmodifiableList(branches);
    private final Map<String, Branch> byName = new LinkedHashMap<>();

    public Branch addBranch(String name, int capacity) throws InvalidDataException, DuplicateBranchException {
        if (name == null || name.trim().isEmpty()) {
            throw new InvalidDataException("Nie dodano oddziału: nazwa jest pusta.");
        }
        String branchName = name.trim();
        if (byName.containsKey(branchName)) {
            throw new DuplicateBranchException("Oddział \"" + branchName + "\" już istnieje.");
        }
        if (capacity <= 0) {
            throw new InvalidDataException("Nie dodano oddziału \"" + branchName + "\": pojemność musi być dodatnia.");
        }
        Branch branch = new Branch(branchName, capacity);
        byName.put(branchName, branch);
        branches.add(branch);
        return branch;
    }

    public Branch removeBranch(String name) throws InvalidDataException, BranchNotFoundException {
        if (name == null || name.trim().isEmpty()) {
            throw new InvalidDataException("Nie usunięto oddziału: brak nazwy.");
        }
        String branchName = name.trim();
        Branch removed = byName.remove(branchName);
        if (removed == null) {
            throw new BranchNotFoundException("Nie znaleziono oddziału \"" + branchName + "\".");
        }
        branches.remove(removed);
        return removed;
    }

    public List<Branch> findEmpty() {
        List<Branch> empty = new ArrayList<>();
        for (Branch branch : branches) {
            if (branch.isEmpty()) {
                empty.add(branch);
            }
        }
        return empty;
    }

    /** Nazwa oddziału i procentowe zapełnienie. */
    public void summary() {
        System.out.println("=== Sieć sklepów (" + branches.size() + " oddziałów) ===");
        if (branches.isEmpty()) {
            System.out.println("Brak oddziałów.");
            return;
        }
        for (Branch branch : branches) {
            System.out.printf("Oddział: %s — zapełnienie %d/%d (%.1f%%)%n",
                    branch.getBranchName(),
                    branch.size(),
                    branch.getMaxCapacity(),
                    branch.fillRatio() * 100.0);
        }
    }

    /**
     * Przenosi laptop między oddziałami. Najpierw sprawdza miejsce i duplikat w celu,
     * dopiero potem zdejmuje laptop ze źródła — nieudany transfer nic nie usuwa.
     */
    public void transferLaptop(Laptop laptop, String fromBranch, String toBranch)
            throws InvalidDataException, BranchNotFoundException, LaptopNotFoundException,
            DuplicateLaptopException, BranchFullException {
        if (laptop == null) {
            throw new InvalidDataException("Nie przeniesiono laptopa: brak obiektu.");
        }
        Branch source = requireBranch(fromBranch, "źródłowego");
        Branch destination = requireBranch(toBranch, "docelowego");
        if (source == destination) {
            throw new InvalidDataException("Nie przeniesiono laptopa: źródło i cel to ten sam oddział.");
        }
        if (!source.contains(laptop)) {
            throw new LaptopNotFoundException("Nie przeniesiono laptopa: nie ma go w oddziale \""
                    + source.getBranchName() + "\".");
        }
        if (destination.contains(laptop)) {
            throw new DuplicateLaptopException("Nie przeniesiono laptopa: w \"" + destination.getBranchName()
                    + "\" jest już egzemplarz o tym samym modelu, marce i wieku.");
        }
        if (destination.isFull()) {
            throw new BranchFullException("Nie przeniesiono laptopa: oddział \"" + destination.getBranchName()
                    + "\" jest pełny (" + destination.size() + "/" + destination.getMaxCapacity() + ").");
        }
        Laptop moved = source.take(laptop);
        destination.attach(moved);
    }

    public double totalValue() {
        double sum = 0;
        for (Branch branch : branches) {
            sum += branch.totalValue();
        }
        return sum;
    }

    public Branch getBranch(String name) {
        if (name == null) {
            return null;
        }
        return byName.get(name.trim());
    }

    /**
     * Ta sama lista, którą ogląda tabela oddziałów.
     * {@link #sortByCurrentLoad()} sortuje ją w miejscu.
     */
    public List<Branch> getBranches() {
        return branchView;
    }

    /** Malejąco po aktualnym zapełnieniu, a przy remisie alfabetycznie po nazwie. */
    public void sortByCurrentLoad() {
        branches.sort(Comparator.comparingDouble(Branch::fillRatio).reversed()
                .thenComparing(Branch::getBranchName));
    }

    private Branch requireBranch(String name, String role) throws InvalidDataException, BranchNotFoundException {
        if (name == null || name.trim().isEmpty()) {
            throw new InvalidDataException("Nie przeniesiono laptopa: brak nazwy oddziału " + role + ".");
        }
        Branch branch = byName.get(name.trim());
        if (branch == null) {
            throw new BranchNotFoundException("Nie przeniesiono laptopa: brak oddziału \"" + name.trim() + "\".");
        }
        return branch;
    }
}

public class StoreFacade {

    private final List<ContactRequest> contactRequests = new ArrayList<>();

    public StoreFacade(StoreManager manager) throws InvalidDataException {

    }

    public void addLaptop(Branch branch, Laptop laptop)
            throws InvalidDataException, BranchNotFoundException, DuplicateLaptopException, BranchFullException {
        if (laptop == null) {
            throw new InvalidDataException("Nie wskazano laptopa.");
        }
        requireBranch(branch).addLaptop(laptop);
    }

    public void removeLaptop(Branch branch, Laptop laptop)
            throws InvalidDataException, BranchNotFoundException, LaptopNotFoundException {
        requireBranch(branch).removeLaptop(laptop);
    }

    /**
     * Jedno wywołanie sortuje oddziały po aktualnym zapełnieniu.
     * Tabela trzyma wynik {@link #getBranches()}, więc nie kopiujemy kolekcji.
     */
    public void sortBranchesByCurrentLoad() {
        manager.sortByCurrentLoad();
    }

    public List<Laptop> filterLaptops(Branch branch, String text, LaptopCondition condition) {
        if (branch == null) {
            return new ArrayList<>();
        }
        String query = text == null ? "" : text.trim().toLowerCase(Locale.ROOT);
        List<Laptop> result = new ArrayList<>();
        for (Laptop laptop : branch.getLaptops()) {
            boolean textMatches = query.isEmpty()
                    || laptop.getModel().toLowerCase(Locale.ROOT).contains(query)
                    || laptop.getBrand().toLowerCase(Locale.ROOT).contains(query);
            boolean conditionMatches = condition == null || laptop.getCondition() == condition;
            if (textMatches && conditionMatches) {
                result.add(laptop);
            }
        }
        return result;
    }

    public void reserveLaptop(Branch branch, Laptop laptop)
            throws InvalidDataException, BranchNotFoundException, LaptopNotFoundException {
        Branch stored = requireBranch(branch);
        Laptop found = requireLaptop(stored, laptop);
        if (found.getCondition() != LaptopCondition.WORKING) {
            throw new InvalidDataException("Rezerwacja jest możliwa tylko dla laptopów w stanie Sprawny.");
        }
        stored.changeCondition(found, LaptopCondition.RESERVED);
    }

    public void changeCondition(Branch branch, Laptop laptop, LaptopCondition condition)

        requireBranch(branch).changeCondition(laptop, condition);
    }

    public void changePrice(Branch branch, Laptop laptop, double price)
            throws InvalidDataException, BranchNotFoundException, LaptopNotFoundException, InvalidPriceException {
        requireBranch(branch).changePrice(laptop, price);
    }

    public void requestContact(Branch branch, Laptop laptop, String clientName, String message)
            throws InvalidDataException, BranchNotFoundException, LaptopNotFoundException {
        Branch stored = requireBranch(branch);
        Laptop found = requireLaptop(stored, laptop);
        if (clientName == null || clientName.trim().isEmpty()) {
            throw new InvalidDataException("Podaj imię i nazwisko.");
        }
        if (message == null || message.trim().isEmpty()) {
            throw new InvalidDataException("Wpisz treść prośby o kontakt.");
        }
        contactRequests.add(new ContactRequest(
                clientName.trim(),
                stored.getBranchName(),
                found.getBrand() + " " + found.getModel() + " (" + found.getSerialNumber() + ")",
                message.trim()));
    }

    public List<ContactRequest> getContactRequests() {
        return Collections.unmodifiableList(contactRequests);
    }

    private Branch requireBranch(Branch branch) throws InvalidDataException, BranchNotFoundException {
        if (branch == null) {
            throw new InvalidDataException("Wybierz oddział.");
        }
        Branch stored = manager.getBranch(branch.getBranchName());
        if (stored == null) {
            throw new BranchNotFoundException("Nie znaleziono oddziału \"" + branch.getBranchName() + "\".");
        }
        return stored;
    }

    private Laptop requireLaptop(Branch branch, Laptop laptop)
            throws InvalidDataException, LaptopNotFoundException {
        if (laptop == null) {
            throw new InvalidDataException("Wybierz laptop.");
        }
        for (Laptop stored : branch.getLaptops()) {
            if (stored.equals(laptop)) {
                return stored;
            }
        }
        throw new LaptopNotFoundException("Brak laptopa w oddziale \"" + branch.getBranchName() + "\".");
    }
}

package lab3.ui;

import lab3.exception.InvalidDataException;

import javax.swing.table.AbstractTableModel;
import java.lang.reflect.InvocationTargetException;
import java.lang.reflect.Method;
import java.util.ArrayList;
import java.util.List;

/**
 * Model tabeli oparty na refleksji. Te same kolumny opisane nazwami getterów
 * obsługują oddziały, laptopy i prośby o kontakt.
 */
public class GenericTableModel<T> extends AbstractTableModel {
    private List<T> rows;
    private final String[] columnNames;
    private final Method[] getters;

    public GenericTableModel(Class<T> type, List<T> rows, String[] columnNames, String[] properties)
            throws InvalidDataException {
        if (type == null || columnNames == null || properties == null || columnNames.length != properties.length) {
            throw new InvalidDataException("Niepoprawna definicja kolumn tabeli.");
        }
        this.rows = rows == null ? new ArrayList<>() : rows;
        this.columnNames = columnNames.clone();
        this.getters = new Method[properties.length];
        for (int i = 0; i < properties.length; i++) {
            getters[i] = findGetter(type, properties[i]);
        }
    }

    public void setRows(List<T> rows) {
        this.rows = rows == null ? new ArrayList<>() : rows;
        fireTableDataChanged();
    }

    public void refresh() {
        fireTableDataChanged();
    }

    public T getRow(int index) {
        return rows.get(index);
    }

    @Override
    public int getRowCount() {
        return rows.size();
    }

    @Override
    public int getColumnCount() {
        return columnNames.length;
    }

    @Override
    public String getColumnName(int column) {
        return columnNames[column];
    }

    @Override
    public boolean isCellEditable(int rowIndex, int columnIndex) {
        return false;
    }

    @Override
    public Object getValueAt(int rowIndex, int columnIndex) {
        try {
            return getters[columnIndex].invoke(rows.get(rowIndex));
        } catch (IllegalAccessException | InvocationTargetException ex) {
            return "";
        }
    }

    private static Method findGetter(Class<?> type, String property) throws InvalidDataException {
        if (property == null || property.isEmpty()) {
            throw new InvalidDataException("Nazwa właściwości jest pusta.");
        }
        String suffix = Character.toUpperCase(property.charAt(0)) + property.substring(1);
        try {
            return type.getMethod("get" + suffix);
        } catch (NoSuchMethodException notFound) {
            try {
                return type.getMethod(property);
            } catch (NoSuchMethodException again) {
                throw new InvalidDataException(
                        "Brak właściwości \"" + property + "\" w klasie " + type.getSimpleName() + ".");
            }
        }
    }
}

        if (!accepted(parent, form, "add new laptop")) {
            return null;
        }
        try {
            return new Laptop(
                    model.getText(),
                    brand.getText(),
                    (LaptopCondition) condition.getSelectedItem(),
                    Parsers.integer(year.getText(), "Rok produkcji"),
                    Parsers.decimal(price.getText(), "Cena"),
                    Parsers.integer(ram.getText(), "RAM"),
                    processor.getText(),
                    Parsers.integer(battery.getText(), "Bateria"));
        } catch (InvalidPriceException | InvalidDataException ex) {
            error(parent, ex.getMessage());
            return null;
        }
    }

    static LaptopCondition condition(Component parent, LaptopCondition current) {
        JComboBox<LaptopCondition> box = new JComboBox<>(LaptopCondition.values());
        box.setSelectedItem(current);
        if (!accepted(parent, box, "edit condition")) {
            return null;
        }
        return (LaptopCondition) box.getSelectedItem();
    }

    static String price(Component parent, double current) {
        Object value = JOptionPane.showInputDialog(parent, "Nowa cena (zł)", "edit price",
                JOptionPane.QUESTION_MESSAGE, null, null, current);
        if (value == null) {
            return null;
        }
        return value.toString();
    }

    static String[] contact(Component parent) {
        JTextField name = new JTextField();
        JTextArea message = new JTextArea(4, 24);

        form.add(new JLabel("Imię i nazwisko"));

        form.add(new JLabel("Wiadomość"));

        }

    }

    private static JPanel form() {
        JPanel form = new JPanel(new GridLayout(0, 2, 6, 6));
        return form;
    }

    private static boolean accepted(Component parent, JComponent form, String title) {
        int result = JOptionPane.showConfirmDialog(parent, form, title,
                JOptionPane.OK_CANCEL_OPTION, JOptionPane.PLAIN_MESSAGE);
        return result == JOptionPane.OK_OPTION;
    }

import lab3.exception.InvalidDataException;

/**
 * Wspólny widok dwóch tabel: oddziały i laptopy wybranego oddziału,
 * razem z polem filter i listą condition.
 */
final class CatalogView {
    final GenericTable<Branch> branches;
    final GenericTable<Laptop> laptops;
    final JTextField filterField;
    final JComboBox<Object> conditionBox;
    private final StoreFacade facade;
    private Branch selected;

    CatalogView(StoreFacade facade) throws InvalidDataException {
        this.facade = facade;
        branches = GenericTable.branches(facade.getBranches());
        laptops = GenericTable.laptops(new ArrayList<Laptop>());
        filterField = new JTextField(18);
        filterField.setToolTipText("Nazwa modelu lub marki. Enter filtruje laptopy wybranego oddziału.");
        conditionBox = new JComboBox<>();
        conditionBox.addItem("Wszystkie");
        for (LaptopCondition condition : LaptopCondition.values()) {
            conditionBox.addItem(condition);
        }
        conditionBox.setToolTipText("Zostaw tylko laptopy w wybranym stanie.");
        branches.onSelection(new java.util.function.Consumer<Branch>() {
            @Override
            public void accept(Branch branch) {
                selected = branch;
                applyFilter();
            }
        });
        filterField.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent event) {
                applyFilter();
            }
        });
        conditionBox.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent event) {
                applyFilter();
            }
        });
        selectFirstBranch();
    }

    Branch getSelectedBranch() {
        return selected;
    }

    Laptop getSelectedLaptop() {
        return laptops.getSelected();
    }

    void selectFirstBranch() {
        if (branches.getRowCount() > 0) {
            branches.selectRow(0);
        }
    }

    void selectBranch(String name) {
        for (int i = 0; i < branches.getRowCount(); i++) {
            if (name.equals(branches.getRow(i).getBranchName())) {
                selected = branches.getRow(i);
                branches.selectRow(i);
                applyFilter();
                return;
            }
        }
    }

    void reloadBranchesKeepingSelection() {
        String preferredName = selected == null ? null : selected.getBranchName();
        branches.refresh();
        if (branches.getRowCount() == 0) {
            selected = null;
            applyFilter();
            return;
        }
        int index = 0;
        if (preferredName != null) {
            for (int i = 0; i < branches.getRowCount(); i++) {
                if (preferredName.equals(branches.getRow(i).getBranchName())) {
                    index = i;
                    break;
                }
            }
        }
        branches.selectRow(index);
    }

    void applyFilter() {
        laptops.setRows(facade.filterLaptops(selected, filterField.getText(), selectedCondition()));
    }

    private LaptopCondition selectedCondition() {
        Object item = conditionBox.getSelectedItem();
        if (item instanceof LaptopCondition) {
            return (LaptopCondition) item;
        }
        return null;
    }
}

import java.io.ByteArrayOutputStream;
import java.io.PrintStream;
import java.nio.charset.StandardCharsets;

public final class OutputCapture {
    private OutputCapture() {
    }

    public static String stdout(Runnable action) {
        PrintStream original = System.out;
        ByteArrayOutputStream buffer = new ByteArrayOutputStream();
        try {
            System.setOut(new PrintStream(buffer, true, StandardCharsets.UTF_8));
            action.run();
            return buffer.toString(StandardCharsets.UTF_8);
        } finally {
            System.setOut(original);
        }
    }
}

package lab3.support;

import javax.swing.SwingUtilities;
import java.util.concurrent.atomic.AtomicReference;

public final class Edt {
    private Edt() {
    }

    public static void run(Action action) throws Exception {
        if (SwingUtilities.isEventDispatchThread()) {
            action.run();
            return;
        }
        AtomicReference<Exception> failure = new AtomicReference<>();
        SwingUtilities.invokeAndWait(() -> {
            try {
                action.run();
            } catch (Exception ex) {
                failure.set(ex);
            }
        });
        if (failure.get() != null) {
            throw failure.get();
        }
    }

    @FunctionalInterface
    public interface Action {
        void run() throws Exception;
    }
}

package lab3.ui;

import lab3.data.DataGenerator;
import lab3.facade.StoreFacade;
import lab3.model.LaptopCondition;
import lab3.support.Edt;
import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;

import javax.swing.JButton;
import java.awt.Component;
import java.awt.Container;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertNotNull;
import static org.junit.jupiter.api.Assertions.assertNull;

@DisplayName("Aplikacja")
public class ApplicationFrameTest {
    @Test
    void loginSwitchesBetweenAdminAndClient() throws Exception {
        StoreFacade facade = new StoreFacade(DataGenerator.createSample());
        Edt.run(() -> {
            ApplicationFrame frame = new ApplicationFrame(facade);
            assertEquals("login", frame.visibleCard());
            assertNotNull(findButton(frame.adminPanel, "add branch"));
            assertNotNull(findButton(frame.adminPanel, "remove branch"));
            assertNotNull(findButton(frame.adminPanel, "add new laptop"));
            assertNotNull(findButton(frame.adminPanel, "remove laptop"));
            assertNotNull(findButton(frame.adminPanel, "sort branches by current load"));
            assertNull(findButton(frame.adminPanel, "reserve"));
            assertNotNull(findButton(frame.clientPanel, "reserve"));
            assertNotNull(findButton(frame.clientPanel, "request contact"));
            assertNull(findButton(frame.clientPanel, "add branch"));
            assertNull(findButton(frame.clientPanel, "remove laptop"));

            frame.loginPanel.username.setText("admin");
            frame.loginPanel.password.setText("admin");
            frame.loginPanel.loginButton.doClick();
            assertEquals("admin", frame.visibleCard());
            findButton(frame.adminPanel, "Wyloguj").doClick();
            assertEquals("login", frame.visibleCard());
            assertEquals("", new String(frame.loginPanel.password.getPassword()));

            frame.loginPanel.username.setText("klient");
            frame.loginPanel.password.setText("klient");
            frame.loginPanel.loginButton.doClick();
            assertEquals("client", frame.visibleCard());
            frame.clientPanel.catalog.laptops.selectRow(0);
            frame.clientPanel.reserveButton.doClick();
            assertEquals(LaptopCondition.RESERVED,
                    frame.clientPanel.catalog.getSelectedBranch().getLaptops().get(0).getCondition());
            findButton(frame.clientPanel, "Wyloguj").doClick();
            assertEquals("login", frame.visibleCard());
            frame.dispose();
        });
    }

    private static JButton findButton(Component root, String text) {
        if (root instanceof JButton) {
            JButton button = (JButton) root;
            if (text.equals(button.getText())) {
                return button;
            }
        }
        if (root instanceof Container) {
            Component[] children = ((Container) root).getComponents();
            for (Component child : children) {
                JButton found = findButton(child, text);
                if (found != null) {
                    return found;
                }
            }
        }
        return null;
    }
}
